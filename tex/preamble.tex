% --- 1. PAGE GEOMETRY (spec B1, B2) ----------------------------------------
% Trim 8.5 x 11 in. Text block 6.50 x 9.35 in, split into a 4.60 in main
% column, a 0.25 in gutter and a 1.65 in side column.
% 0.75 + 4.60 + 0.25 + 1.65 + 1.25 = 8.50 in.
%
% DEVIATION FROM SPEC B1, deliberate and documented in docs/build.md:
% the margins do not mirror. `asymmetric` fixes the narrow margin on the left
% and the side column on the right for every page. Two reasons.
%   1. Correctness. A mirrored layout has to decide which way a full-width
%      element breaks out, and that decision depends on the page a box will
%      eventually land on. LaTeX builds boxes before it knows that, so the
%      answer is read from the previous compilation pass and is wrong whenever
%      pagination shifts. The failure mode is silent: text runs off the trim
%      edge. On a 240-page book with hundreds of full-width components that is
%      not a risk worth carrying.
%   2. Fit. This is a digital-first PDF read on screen and printed one-sided on
%      school printers. Mirrored margins buy nothing without a binding, and
%      they make a one-sided printout look broken. The spec makes the same
%      trade in B6 when it refuses to force recto-only chapter openers.
% Alternating running heads are kept, because fancyhdr resolves parity at
% shipout, where it is always correct.
\usepackage{geometry}
\geometry{
  paperwidth      = 8.5in,
  paperheight     = 11in,
  top             = 0.75in,
  bottom          = 0.90in,
  left            = 0.75in,
  right           = 1.25in,
  marginparsep    = 0.25in,
  marginparwidth  = 1.65in,
  includemp       = true,
  twoside         = true,
  asymmetric      = true,
  headheight      = 12pt,
  headsep         = 19.7pt,   % puts the running-head baseline 0.45in from trim
  footskip        = 24.2pt,   % puts the folio baseline 0.55in from trim
}
% Margin material always goes in the side column on the right.

% JetBrains Mono NL is the no-ligature cut. Spec B3: ligatures MUST be off in
% code so a beginner sees the characters they have to type (!= not ≠).
% No Scale= here on purpose. Spec B4 sets code at a true 9.5 pt because the
% 80-column guarantee depends on it: JetBrains Mono advances 0.6 em per
% character, so 80 x 0.6 x 9.5 pt = 456 pt = 6.33 in, which fits the 6.50 in
% text block. Matching x-heights would shrink that to ~8.2 pt and quietly
% invalidate the measurement the whole code layout rests on.
\setmonofont{JetBrains Mono NL}[
  UprightFont    = *-Regular,
  ItalicFont     = *-Italic,
  BoldFont       = *-Bold,
  BoldItalicFont = *-BoldItalic,
  Ligatures      = NoCommon,
]

% 9.1 Margin note and margin definition ......................................
% \marginpar, not the marginnote package. marginnote picks its side from
% \if@twoside and \ifodd\value{page} directly, so it lands in the left margin on
% even pages however \@mparswitch is set; \marginpar honours \@mparswitchfalse
% and is therefore deterministic under the asymmetric geometry above.
% The cost is that \marginpar is illegal inside a box, so every boxed component
% below rebinds these two commands to inline-only versions.
\setlength{\marginparpush}{10pt}
% LaTeX only ever pushes a colliding \marginpar downward, so a definition
% declared near the foot of a page runs off the bottom of the paper. marginfix
% collects the margin material for each page and redistributes it, moving what
% genuinely does not fit onto the next page instead of off the trim edge.
\usepackage{marginfix}
